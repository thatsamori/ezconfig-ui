  <action>
Apply the same blur-to-commit pattern to FloatArrayInput component.

FloatArrayInput manages an array of float values. Each input in the array should:
1. Use local string state for its value
2. Only commit on blur or Enter

Implementation:
```tsx
const [localValues, setLocalValues] = useState(value.map(String));

useEffect(() => {
  setLocalValues(value.map(String));
}, [value]);

const handleChange = (index: number, newValue: string) => {
  setLocalValues(prev => {
    const updated = [...prev];
    updated[index] = newValue;
    return updated;
  });
};

const handleBlur = () => {
  const parsed = localValues.map(v => parseFloat(v));
  if (parsed.every(n => !isNaN(n))) {
    onChange(parsed);
  } else {
    setLocalValues(value.map(String)); // Revert
  }
};
```
