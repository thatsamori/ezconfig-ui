     ```typescript
     const zip = new JSZip();

     // Add manifest
     zip.file("manifest.json", JSON.stringify({
       title: preset.manifest.title,
       description: preset.manifest.description
     }, null, 2));

     // Add Character configs
     for (const [category, values] of Object.entries(data.character)) {
       const entries = Object.entries(values).map(([k, v]) => ({ [k]: v }));
       zip.file(`Character/${category}.json`, JSON.stringify(entries, null, 2));
     }

     // Add Weapon configs
     for (const [weaponName, categories] of Object.entries(data.weapons)) {
       for (const [category, values] of Object.entries(categories)) {
         const entries = Object.entries(values).map(([k, v]) => ({ [k]: v }));
         zip.file(`Weapon/${weaponName}/${category}.json`, JSON.stringify(entries, null, 2));
       }
     }
     ```

   ```typescript
   a.download = `${preset.name}.zip`;
   a.click();
   URL.revokeObjectURL(url);
   ```

<task type="auto">
  <name>Task 2: Add import functionality for preset ZIP files</name>
  <files>src/components/presets/PresetsTab.tsx, src/app/api/presets/import/route.ts</files>
  <action>
1. Add Import button in Presets tab header:
   - Place next to "Save Current Config as Preset" button
   - Use hidden file input triggered by button click
   - Accept only .zip files: `accept=".zip"`

2. Create handleImportPreset function (client-side):
   - Read uploaded file with JSZip.loadAsync(file)
   - Validate structure:
     - Must have manifest.json at root
     - Parse manifest, validate title and description exist
   - Extract preset name from manifest title (slugify: lowercase, replace spaces with hyphens)
   - Check for name conflicts with existing presets
   - If conflict, append number: my-preset-2

3. Create POST /api/presets/import endpoint:
   - Accept multipart form data with ZIP file
   - Use JSZip on server to extract and validate
   - Extract to User presets directory using existing folder structure
   - Return created preset info or error

4. Server-side extraction logic:
   ```typescript
   const zip = await JSZip.loadAsync(buffer);

   // Validate manifest exists
   const manifestFile = zip.file("manifest.json");
   if (!manifestFile) throw new Error("Missing manifest.json");

   const manifest = JSON.parse(await manifestFile.async("string"));

   // Generate safe preset name from title
   const name = slugify(manifest.title);
   const presetPath = join(getUserPresetsPath(), name);

   // Extract all files
   await mkdir(presetPath, { recursive: true });

   for (const [path, zipEntry] of Object.entries(zip.files)) {
     if (zipEntry.dir) {
       await mkdir(join(presetPath, path), { recursive: true });
     } else {
       const content = await zipEntry.async("nodebuffer");
       await writeFile(join(presetPath, path), content);
     }
   }
   ```

5. After successful import:
   - Refresh preset list
   - Show success toast with preset name
   - Clear file input

6. Error handling:
   - Invalid ZIP format
   - Missing manifest.json
   - Invalid manifest structure
   - File write errors
